## 三、部署步骤

### Step 0: 准备 (所有 Ubuntu 服务器)

```bash
# 登录每台 VPS
ssh root@108.160.128.34   # VPS1
ssh root@207.148.105.150  # VPS2
ssh root@45.77.179.42     # VPS3

# 每台执行以下命令:

# 1. 更新系统
apt update && apt upgrade -y

# 2. 安装 Docker
curl -fsSL https://get.docker.com | sh
systemctl start docker
systemctl enable docker

# 3. 安装 JDK 8
apt install -y openjdk-8-jdk

# 4. 安装 Maven
apt install -y maven

# 5. 安装 Node.js (仅 VPS1 需要)
curl -fsSL https://deb.nodesource.com/setup_18.x | bash -
apt install -y nodejs

# 6. 开放端口
ufw allow 80/tcp
ufw allow 8080:8090/tcp
ufw allow 3306/tcp
ufw allow 8123/tcp
ufw allow 6379/tcp
ufw allow 2181/tcp
ufw allow 9092/tcp
ufw allow 9876/tcp
ufw allow 10911/tcp
ufw allow 9200/tcp
ufw reload
```

### Step 3: 部署 VPS2 (客户环境 A)

```bash
ssh root@207.148.105.150

# 1. 克隆项目
cd /root
git clone https://github.com/jiashuaidawang-web/AIPerformanceEngineer.git
cd AIPerformanceEngineer

# 2. 构建
mvn clean package -DskipTests

# 3. 启动中间件 (Kafka + ZooKeeper)
docker-compose up -d zookeeper kafka
sleep 30

# 4. 验证中间件
docker ps | grep -E 'zookeeper|kafka'

# 5. 构建 Demo 微服务
cd aipe-demo-microservices
mvn clean package -DskipTests
cd ..

# 6. 启动微服务
mkdir -p /var/log/demo
nohup java -jar aipe-demo-microservices/order-service/target/order-service-1.0.0-SNAPSHOT.jar \
  --server.port=8082 \
  --spring.kafka.bootstrap-servers=localhost:9092 \
  > /var/log/demo/order.log 2>&1 &

nohup java -jar aipe-demo-microservices/product-service/target/product-service-1.0.0-SNAPSHOT.jar \
  --server.port=8081 \
  > /var/log/demo/product.log 2>&1 &

# 7. 配置并启动 Agent
mkdir -p /root/agent
cat > /root/agent/application.yml << 'AGENT'
agent:
  agentId: "agent-vps2-001"
  serverId: "vps2"
  environment: "prod"
  backendUrl: "http://108.160.128.34:8081"
  schedulerPoolSize: 4
  sendTimeoutMs: 5000
  connectors:
    - type: JVM
      enabled: true
      intervalMs: 30000
    - type: LINUX
      enabled: true
      intervalMs: 30000
      properties:
        procPath: /proc
    - type: KAFKA
      enabled: true
      intervalMs: 30000
      properties:
        host: "localhost"
        port: "9092"
        jmxHost: "localhost"
        jmxPort: "9999"
    - type: ZOOKEEPER
      enabled: true
      intervalMs: 30000
      properties:
        host: "localhost"
        port: "2181"
AGENT

nohup java -jar aipe-agent/target/aipe-agent-1.0.0-SNAPSHOT.jar \
  --spring.config.location=/root/agent/application.yml \
  > /var/log/aipe/agent.log 2>&1 &

echo "VPS2 部署完成!"
```
